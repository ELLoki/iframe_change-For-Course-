  // Converte AudioBuffer para WAV
  const audioBufferToWav = (buffer) => {
    const length = buffer.length;
    const numberOfChannels = buffer.numberOfChannels;
    const sampleRate = buffer.sampleRate;
    const arrayBuffer = new ArrayBuffer(44 + length * numberOfChannels * 2);
    const view = new DataView(arrayBuffer);
    
    // WAV header
    const writeString = (offset, string) => {
      for (let i = 0; i < string.length; i++) {
        view.setUint8(offset + i, string.charCodeAt(i));
      }
    };
    
    writeString(0, 'RIFF');
    view.setUint32(4, 36 + length * numberOfChannels * 2, true);
    writeString(8, 'WAVE');
    writeString(12, 'fmt ');
    view.setUint32(16, 16, true);
    view.setUint16(20, 1, true);
    view.setUint16(22, numberOfChannels, true);
    view.setUint32(24, sampleRate, true);
    view.setUint32(28, sampleRate * numberOfChannels * 2, true);
    view.setUint16(32, numberOfChannels * 2, true);
    view.setUint16(34, 16, true);
    writeString(36, 'data');
    view.setUint32(40, length * numberOfChannels * 2, true);
    
    // Audio data
    let offset = 44;
    for (let i = 0; i < length; i++) {
      for (let channel = 0; channel < numberOfChannels; channel++) {
        const sample = Math.max(-1, Math.min(1, buffer.getChannelData(channel)[i]));
        view.setInt16(offset, sample < 0 ? sample * 0x8000 : sample * 0x7FFF, true);
        offset += 2;
      }
    }
    
    return arrayBuffer;
  };

            return;
          }
        } catch (urlError) {
          console.warn('⚠️ Não foi possível verificar origem:', urlError.message);
        }
      }

      showDownloadStatus('🔄 Capturando vídeo...', 'progress');
      
      // Tenta usar captureStream primeiro
      let stream = null;
      try {
        if (!v.captureStream) {
          throw new Error('captureStream não suportado neste navegador');
        }
        
        // Tenta definir crossOrigin antes de capturar (pode não funcionar se já carregado)
        // Nota: crossOrigin só pode ser definido antes de definir src
        try {
          if (!v.crossOrigin && (v.src || v.currentSrc)) {
            console.warn('⚠️ Vídeo já carregado. crossOrigin não pode ser alterado.');
          } else if (!v.crossOrigin) {
            v.crossOrigin = 'anonymous';
            console.log('✅ crossOrigin definido como anonymous');
          }
        } catch (crossOriginError) {
          console.warn('⚠️ Não foi possível definir crossOrigin:', crossOriginError);
        }
        
        stream = v.captureStream();
        // Log apenas uma vez quando conseguir
        if (!downloadAsMP4.streamCaptured) {
          console.log('✅ Stream capturado com sucesso!');
          downloadAsMP4.streamCaptured = true;
        }
      } catch (e) {
        console.error('❌ Erro ao usar captureStream:', e);
        console.error('  - Tipo do erro:', e.name);
        console.error('  - Mensagem:', e.message);
        
        // Verifica se é erro de cross-origin (pode estar na mensagem ou no tipo)
        const isCrossOriginError = e.message && (
          e.message.includes('cross-origin') || 
          e.message.includes('cross origin') ||
          e.message.includes('Cannot capture')
        ) || e.name === 'SecurityError';
        
        if (isCrossOriginError) {
          showDownloadStatus('⚠️ Vídeo cross-origin detectado. Tentando método alternativo...', 'progress');
          
          // Para vídeos cross-origin, não é possível usar captureStream
          // Tenta criar um link de download direto
          const videoUrl = v.src || v.currentSrc;
          if (videoUrl) {
            console.log('🔗 Tentando download direto da URL:', videoUrl);
            
            try {
              // Cria link de download direto
              const a = innerDoc.createElement('a');
              a.href = videoUrl;
              a.download = `video_${Date.now()}.mp4`;
              a.target = '_blank';
              a.style.display = 'none';
              innerDoc.body.appendChild(a);
              a.click();
              
              setTimeout(() => {
                innerDoc.body.removeChild(a);
              }, 100);
              
              showDownloadStatus('✅ Link de download criado! Se não funcionar, o servidor está bloqueando por CORS.', 'info');
              
              // Informa sobre limitações
              setTimeout(() => {
                showDownloadStatus('💡 Dica: Para vídeos cross-origin, use uma extensão de download de vídeo do navegador.', 'info');
              }, 3000);
              
              return;
            } catch (e2) {
              console.error('Erro ao baixar diretamente:', e2);
              throw new Error('Não foi possível baixar o vídeo. Vídeos cross-origin não podem ser capturados devido a restrições de segurança do navegador. Use uma extensão de download de vídeo.');
            }
          } else {
            throw new Error('URL do vídeo não disponível. Vídeos cross-origin não podem ser capturados devido a restrições de segurança.');
          }
        } else {
          throw e;
        }
      }
      
      // Verifica codecs disponíveis
      const mimeTypes = [
        'video/webm;codecs=vp9',
        'video/webm;codecs=vp8',
        'video/webm',
        'video/mp4'
      ];
      
      let selectedMimeType = 'video/webm';
      for (const mimeType of mimeTypes) {
        if (MediaRecorder.isTypeSupported(mimeType)) {
          selectedMimeType = mimeType;
          break;
        }
      }
      
      const mediaRecorder = new MediaRecorder(stream, {
        mimeType: selectedMimeType
      });
      
      const chunks = [];
      
      mediaRecorder.ondataavailable = (e) => {
        if (e.data && e.data.size > 0) {
          chunks.push(e.data);
          showDownloadStatus(`📥 Gravando... ${chunks.length} chunks`, 'progress');
        }
      };
      
      return new Promise((resolve, reject) => {
        mediaRecorder.onstop = async () => {
          try {
            if (chunks.length === 0) {
              throw new Error('Nenhum dado foi gravado');
            }
            
            showDownloadStatus('🔄 Processando vídeo...', 'progress');
            const blob = new Blob(chunks, { type: selectedMimeType });
            
            // Converte para MP4 se necessário
            const mp4Blob = await convertToMP4(blob);
            const extension = selectedMimeType.includes('mp4') ? 'mp4' : 'webm';
            
            const url = URL.createObjectURL(mp4Blob);
            const a = innerDoc.createElement('a');
            a.href = url;
            a.download = `video_${Date.now()}.${extension}`;
            innerDoc.body.appendChild(a); // Necessário para alguns navegadores
            a.click();
            innerDoc.body.removeChild(a);
            
            URL.revokeObjectURL(url);
            showDownloadStatus('✅ Download concluído!', 'success');
            resolve();
          } catch (e) {
            reject(e);
          }
        };
        
        mediaRecorder.onerror = (e) => {
          reject(new Error('Erro ao gravar: ' + e.error));
        };
        
        // Salva estado do vídeo
        const currentTime = v.currentTime;
        const wasPlaying = !v.paused;
        const playbackRate = v.playbackRate;
        
        // Configura para gravação
        v.currentTime = 0;
        v.playbackRate = 1;
        
        // Inicia gravação
        mediaRecorder.start(1000); // Coleta dados a cada 1 segundo
        
        // Toca o vídeo para capturar
        v.play().then(() => {
          showDownloadStatus('▶️ Gravando vídeo...', 'progress');
          
          // Para quando o vídeo terminar
          v.onended = () => {
            if (mediaRecorder.state === 'recording') {
              mediaRecorder.stop();
              // Restaura estado
              v.currentTime = currentTime;
              v.playbackRate = playbackRate;
              if (!wasPlaying) v.pause();
            }
          };
          
          // Timeout baseado na duração do vídeo
          const duration = v.duration && !isNaN(v.duration) ? v.duration * 1000 : 60000;
          const timeout = Math.min(duration, 300000); // Máximo 5 minutos
          
          setTimeout(() => {
            if (mediaRecorder.state === 'recording') {
              mediaRecorder.stop();
              // Restaura estado
              v.currentTime = currentTime;
              v.playbackRate = playbackRate;
              if (!wasPlaying) v.pause();
            }
          }, timeout);
        }).catch(reject);
      });
      
    } catch (e) {
      console.error('Erro ao baixar MP4:', e);
      showDownloadStatus('❌ Erro ao baixar MP4: ' + e.message, 'error');
    }
  };

  // Extrai áudio de um arquivo de vídeo
  const extractAudioFromFile = async (file, videoPath) => {
    let videoUrl = null;
    let tempVideo = null;
    
    try {
      showDownloadStatus('🔄 Carregando arquivo de vídeo...', 'progress');
      
      // Cria URL temporária para o arquivo
      videoUrl = URL.createObjectURL(file);
      
      // Carrega o vídeo em um elemento de vídeo
      tempVideo = innerDoc.createElement('video');
      tempVideo.src = videoUrl;
      tempVideo.crossOrigin = 'anonymous';
      tempVideo.preload = 'auto';
      tempVideo.style.display = 'none';
      innerDoc.body.appendChild(tempVideo);
      
      // Aguarda o vídeo carregar os metadados
      await new Promise((resolve, reject) => {
        const timeout = setTimeout(() => {
          reject(new Error('Timeout ao carregar vídeo'));
        }, 30000); // 30 segundos timeout
        
        tempVideo.onloadedmetadata = () => {
          clearTimeout(timeout);
          resolve();
        };
        
        tempVideo.onerror = (e) => {
          clearTimeout(timeout);
          reject(new Error('Erro ao carregar vídeo: ' + (tempVideo.error?.message || 'Erro desconhecido')));
        };
        
        tempVideo.load();
      });
      
      showDownloadStatus('🔄 Extraindo áudio do vídeo...', 'progress');
      
      // Método simplificado: usa MediaRecorder para capturar o áudio
      // Cria um stream a partir do elemento de vídeo
      let audioStream = null;
      
      try {
        // Tenta usar captureStream (funciona em navegadores modernos)
        if (tempVideo.captureStream) {
          const stream = tempVideo.captureStream();
          const audioTracks = stream.getAudioTracks();
          
          if (audioTracks.length > 0) {
            audioStream = new MediaStream(audioTracks);
          } else {
            throw new Error('Vídeo não possui faixa de áudio');
          }
        } else {
          // Fallback: usa Web Audio API
          const audioContext = new (window.AudioContext || window.webkitAudioContext)();
          const source = audioContext.createMediaElementSource(tempVideo);
          const destination = audioContext.createMediaStreamDestination();
          source.connect(destination);
          audioStream = destination.stream;
        }
      } catch (streamError) {
        throw new Error('Não foi possível capturar áudio do vídeo: ' + streamError.message);
      }
      
      // Verifica codecs disponíveis para áudio
      const mimeTypes = [
        'audio/webm;codecs=opus',
        'audio/webm',
        'audio/mp4',
        'audio/ogg'
      ];
      
      let selectedMimeType = 'audio/webm';
      for (const mimeType of mimeTypes) {
        if (MediaRecorder.isTypeSupported(mimeType)) {
          selectedMimeType = mimeType;
          break;
        }
      }
      
      const mediaRecorder = new MediaRecorder(audioStream, {
        mimeType: selectedMimeType
      });
      
      const audioChunks = [];
      
      mediaRecorder.ondataavailable = (e) => {
        if (e.data && e.data.size > 0) {
          audioChunks.push(e.data);
          const progress = ((audioChunks.length * 100) / Math.ceil((tempVideo.duration || 60) / 1)).toFixed(0);
          showDownloadStatus(`🎵 Gravando áudio... ${progress}%`, 'progress');
        }
      };
      
      // Inicia a gravação com intervalo maior para melhor performance
      mediaRecorder.start(500); // Coleta dados a cada 500ms (mais frequente para melhor qualidade)
      
      // OTIMIZAÇÃO: Acelera a reprodução do vídeo para processar mais rápido
      // Velocidade de 4x reduz o tempo de processamento de 30min para ~7.5min
      // Velocidade de 8x reduz para ~3.75min (mas pode ter problemas de qualidade)
      // Nota: Velocidades muito altas (8x+) podem causar perda de qualidade ou erros
      let playbackSpeed = 4; // 4x mais rápido (padrão)
      
      // Tenta detectar o tamanho do arquivo e ajustar a velocidade
      // Arquivos menores podem ser processados mais rápido
      if (file.size) {
        const fileSizeMB = file.size / 1024 / 1024;
        if (fileSizeMB < 100) {
          playbackSpeed = 6; // 6x para arquivos menores
        } else if (fileSizeMB > 500) {
          playbackSpeed = 3; // 3x para arquivos muito grandes (mais estável)
        }
      }
      
      // Limita a velocidade máxima para evitar problemas
      playbackSpeed = Math.min(playbackSpeed, 8);
      tempVideo.playbackRate = playbackSpeed;
      tempVideo.volume = 0;
      tempVideo.muted = true; // Garante que está mudo
      
      // Calcula tempo estimado
      const originalDuration = tempVideo.duration || 60;
      const estimatedTime = (originalDuration / playbackSpeed / 60).toFixed(1);
      showDownloadStatus(`▶️ Processando áudio em ${playbackSpeed}x (estimado: ~${estimatedTime} min)...`, 'progress');
      
      await tempVideo.play();
      
      // Monitora progresso em tempo real
      let lastProgressUpdate = 0;
      const progressInterval = setInterval(() => {
        if (tempVideo.duration && tempVideo.currentTime) {
          const progress = Math.min(100, Math.floor((tempVideo.currentTime / tempVideo.duration) * 100));
          const elapsed = (tempVideo.currentTime / playbackSpeed).toFixed(1);
          const remaining = ((tempVideo.duration - tempVideo.currentTime) / playbackSpeed).toFixed(1);
          showDownloadStatus(`🎵 Extraindo áudio... ${progress}% (${elapsed}s / ~${remaining}s restantes)`, 'progress');
        }
      }, 500); // Atualiza a cada 500ms
      
      // Aguarda o vídeo terminar
      await new Promise((resolve, reject) => {
        const timeout = setTimeout(() => {
          clearInterval(progressInterval);
          if (mediaRecorder.state === 'recording') {
            mediaRecorder.stop();
          }
          reject(new Error('Timeout ao processar áudio'));
        }, (originalDuration / playbackSpeed) * 1000 + 10000); // Duração acelerada + 10 segundos
        
        tempVideo.onended = () => {
          clearTimeout(timeout);
          clearInterval(progressInterval);
          if (mediaRecorder.state === 'recording') {
            mediaRecorder.stop();
          }
          resolve();
        };
        
        tempVideo.onerror = (e) => {
          clearTimeout(timeout);
          clearInterval(progressInterval);
          if (mediaRecorder.state === 'recording') {
            mediaRecorder.stop();
          }
          reject(new Error('Erro ao processar vídeo: ' + (tempVideo.error?.message || 'Erro desconhecido')));
        };
      });
      
      // Para o vídeo e restaura configurações
      tempVideo.pause();
      tempVideo.currentTime = 0;
      tempVideo.playbackRate = 1;
      tempVideo.volume = 1;
      tempVideo.muted = false;
      
      showDownloadStatus('🔄 Finalizando gravação...', 'progress');
      
      // Aguarda a gravação terminar (com timeout maior para arquivos grandes)
      await new Promise((resolve, reject) => {
        let resolved = false;
        
        mediaRecorder.onstop = () => {
          if (!resolved) {
            resolved = true;
            resolve();
          }
        };
        
        mediaRecorder.onerror = (e) => {
          if (!resolved) {
            resolved = true;
            reject(new Error('Erro ao gravar áudio: ' + e.error));
          }
        };
        
        // Timeout de segurança maior para arquivos grandes
        setTimeout(() => {
          if (!resolved && mediaRecorder.state === 'recording') {
            resolved = true;
            mediaRecorder.stop();
            // Aguarda mais um pouco para o stop processar
            setTimeout(() => resolve(), 1000);
          }
        }, 10000); // 10 segundos para finalizar
      });
      
      if (audioChunks.length === 0) {
        throw new Error('Nenhum áudio foi gravado');
      }
      
      showDownloadStatus(`🔄 Processando ${audioChunks.length} chunks de áudio...`, 'progress');
      
      // OTIMIZAÇÃO: Processa em chunks para não travar a UI
      // Cria o blob de áudio
      const audioBlob = new Blob(audioChunks, { type: selectedMimeType });
      
      showDownloadStatus('🔄 Convertendo para WAV...', 'progress');
      
      // Tenta converter para WAV usando Web Audio API
      let wavBlob = audioBlob;
      try {
        showDownloadStatus('🔄 Decodificando áudio...', 'progress');
        
        // Processa a conversão de forma assíncrona para não travar a UI
        const arrayBuffer = await audioBlob.arrayBuffer();
        
        // Usa AudioContext normal (OfflineAudioContext não funciona bem com decodeAudioData)
        const audioContext = new (window.AudioContext || window.webkitAudioContext)();
        
        // Decodifica o áudio (isso pode demorar para arquivos grandes)
        const audioBuffer = await audioContext.decodeAudioData(arrayBuffer);
        
        showDownloadStatus('🔄 Convertendo para WAV...', 'progress');
        
        // Processa a conversão em chunks para não bloquear a UI
        // Usa requestIdleCallback se disponível, senão usa setTimeout
        const wavArrayBuffer = await new Promise((resolve) => {
          const convert = () => {
            try {
              const wav = audioBufferToWav(audioBuffer);
              resolve(wav);
            } catch (e) {
              throw e;
            }
          };
          
          // Tenta usar requestIdleCallback para processar quando o navegador estiver ocioso
          if (window.requestIdleCallback) {
            requestIdleCallback(convert, { timeout: 1000 });
          } else {
            // Fallback: processa no próximo frame
            setTimeout(convert, 0);
          }
        });
        
        wavBlob = new Blob([wavArrayBuffer], { type: 'audio/wav' });
        showDownloadStatus('✅ Áudio convertido para WAV!', 'success');
      } catch (convertError) {
        console.warn('Não foi possível converter para WAV, mantendo formato original:', convertError);
        showDownloadStatus('⚠️ Mantendo formato original (conversão WAV falhou)', 'info');
        // Mantém o formato original
      }
      
      // Pega o nome do arquivo sem extensão
      const videoFileName = file.name.replace(/\.[^/.]+$/, '');
      const isWav = wavBlob.type.includes('wav');
      const audioFileName = `${videoFileName}.${isWav ? 'wav' : 'mp3'}`;
      const fileExtension = isWav ? 'wav' : (selectedMimeType.includes('mp4') ? 'm4a' : 'webm');
      
      // Tenta salvar na mesma pasta do vídeo usando File System Access API
      if ('showSaveFilePicker' in window) {
        try {
          // Tenta abrir o seletor de arquivo para salvar na mesma pasta
          const fileHandle = await window.showSaveFilePicker({
            suggestedName: audioFileName,
            types: [{
              description: `Arquivo de áudio ${fileExtension.toUpperCase()}`,
              accept: { [`audio/${fileExtension}`]: [`.${fileExtension}`] }
            }]
          });
          
          // Salva o arquivo
          const writable = await fileHandle.createWritable();
          await writable.write(wavBlob);
          await writable.close();
          
          showDownloadStatus('✅ Áudio extraído e salvo com sucesso na mesma pasta do vídeo!', 'success');
          
          // Limpa recursos
          if (tempVideo && tempVideo.parentNode) {
            innerDoc.body.removeChild(tempVideo);
          }
          if (videoUrl) {
            URL.revokeObjectURL(videoUrl);
          }
          
          return;
        } catch (saveError) {
          if (saveError.name === 'AbortError') {
            showDownloadStatus('⚠️ Operação cancelada pelo usuário', 'info');
            if (tempVideo && tempVideo.parentNode) {
              innerDoc.body.removeChild(tempVideo);
            }
            if (videoUrl) {
              URL.revokeObjectURL(videoUrl);
            }
            return;
          }
          console.warn('Não foi possível usar File System Access API:', saveError);
          // Continua com o método de download tradicional
        }
      }
      
      // Método alternativo: faz download do arquivo
      const audioUrl = URL.createObjectURL(wavBlob);
      const a = innerDoc.createElement('a');
      a.href = audioUrl;
      a.download = `${videoFileName}.${fileExtension}`;
      
      innerDoc.body.appendChild(a);
      a.click();
      innerDoc.body.removeChild(a);
      
      URL.revokeObjectURL(audioUrl);
      
      showDownloadStatus(`✅ Áudio extraído! O arquivo será baixado na pasta de downloads.`, 'success');
      
      // Limpa recursos
      if (tempVideo) {
        innerDoc.body.removeChild(tempVideo);
      }
      if (videoUrl) {
        URL.revokeObjectURL(videoUrl);
      }
      
    } catch (e) {
      console.error('Erro ao extrair áudio:', e);
      showDownloadStatus('❌ Erro ao extrair áudio: ' + e.message, 'error');
      
      // Limpa recursos em caso de erro
      if (tempVideo && tempVideo.parentNode) {
        innerDoc.body.removeChild(tempVideo);
      }
      if (videoUrl) {
        URL.revokeObjectURL(videoUrl);
      }
      
      throw e;
    }
  };
